cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_texture_text_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build texture tools and quality demo" ON)
    option(RENDERER_BUILD_TESTS "Register example checks" ON)
    option(RENDERER_GL_TESTS "Register OpenGL checks" OFF)
    option(RENDERER_VULKAN_TESTS "Register Vulkan checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS text opengl vulkan)
endif()
add_executable(renderer_texture_pack pack.cpp)
target_link_libraries(renderer_texture_pack PRIVATE renderer::core)
set(texture_example_targets renderer_texture_pack)
if(TARGET renderer::text AND (TARGET renderer::opengl OR TARGET renderer::vulkan))
    add_executable(renderer_texture_text_demo main.cpp)
    list(APPEND texture_example_targets renderer_texture_text_demo)
    target_link_libraries(renderer_texture_text_demo PRIVATE renderer::text)
    foreach(backend IN ITEMS opengl vulkan)
        if(TARGET renderer::${backend})
            string(TOUPPER "${backend}" upper)
            target_compile_definitions(renderer_texture_text_demo PRIVATE RENDERER_TEXTURE_${upper}=1)
            target_link_libraries(renderer_texture_text_demo PRIVATE renderer::${backend})
        endif()
    endforeach()
    target_compile_definitions(renderer_texture_text_demo PRIVATE RENDERER_TEXTURE_FONT="${renderer_TEXT_FONT}")
endif()
foreach(target IN LISTS texture_example_targets)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        include("${renderer_EXAMPLE_TEST_HELPER}")
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Tests disabled.\n")
    endif()
endif()
