cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_sdk_consumer LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the SDK consumer example" ON)
    option(RENDERER_BUILD_TESTS "Register example runtime checks" ON)
    option(RENDERER_GL_TESTS "Register checks requiring an OpenGL display" ON)
    option(SDK_CONSUMER_OPENGL "Run a hidden OpenGL frame in addition to CPU APIs" ON)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples are disabled.\n")
        return()
    endif()
    if(SDK_CONSUMER_OPENGL)
        find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core opengl)
    else()
        find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core)
    endif()
else()
    if(TARGET renderer::opengl)
        set(SDK_CONSUMER_OPENGL ON)
    else()
        set(SDK_CONSUMER_OPENGL OFF)
    endif()
endif()
if(SDK_CONSUMER_OPENGL)
    set(renderer_consumer_target renderer::opengl)
else()
    set(renderer_consumer_target renderer::core)
endif()

add_executable(renderer_sdk_consumer main.cpp)
target_link_libraries(renderer_sdk_consumer PRIVATE ${renderer_consumer_target})
set_target_properties(renderer_sdk_consumer PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>"
    RENDERER_EXAMPLE_REQUIRES_OPENGL ${SDK_CONSUMER_OPENGL})
target_compile_definitions(renderer_sdk_consumer PRIVATE
    SDK_CONSUMER_OPENGL=$<BOOL:${SDK_CONSUMER_OPENGL}>)
if(MSVC)
    target_compile_options(renderer_sdk_consumer PRIVATE /W4 /permissive- /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Example tests are disabled.\n")
    endif()
endif()
