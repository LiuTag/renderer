cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_modules_consumer LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the module consumer example" ON)
    option(RENDERER_BUILD_TESTS "Register example runtime checks" ON)
    option(SDK_CONSUMER_ANIMATION "Use the animation extension" ON)
    option(SDK_CONSUMER_INTERACTION "Use the interaction extension" ON)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples are disabled.\n")
        return()
    endif()
else()
    foreach(module IN ITEMS ANIMATION INTERACTION)
        string(TOLOWER "${module}" component)
        if(TARGET renderer::${component})
            set(SDK_CONSUMER_${module} ON)
        else()
            set(SDK_CONSUMER_${module} OFF)
        endif()
    endforeach()
endif()
set(consumer_components core)
set(consumer_libraries renderer::core)
foreach(module IN ITEMS ANIMATION INTERACTION)
    if(SDK_CONSUMER_${module})
        string(TOLOWER "${module}" component)
        list(APPEND consumer_components ${component})
        list(APPEND consumer_libraries renderer::${component})
    endif()
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS ${consumer_components})
endif()
add_executable(renderer_modules_consumer main.cpp)
target_link_libraries(renderer_modules_consumer PRIVATE ${consumer_libraries})
set_target_properties(renderer_modules_consumer PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>"
    RENDERER_EXAMPLE_REQUIRES_OPENGL OFF)
target_compile_definitions(renderer_modules_consumer PRIVATE
    SDK_CONSUMER_ANIMATION=$<BOOL:${SDK_CONSUMER_ANIMATION}>
    SDK_CONSUMER_INTERACTION=$<BOOL:${SDK_CONSUMER_INTERACTION}>)
if(MSVC)
    target_compile_options(renderer_modules_consumer PRIVATE /W4 /permissive- /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Example tests are disabled.\n")
    endif()
endif()
