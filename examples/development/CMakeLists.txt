cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_development_examples LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build development examples" ON)
    option(RENDERER_BUILD_TESTS "Register example runtime checks" ON)
    option(RENDERER_GL_TESTS "Register checks requiring an OpenGL display" ON)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples are disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core opengl
        OPTIONAL_COMPONENTS animation interaction)
endif()
set(development_examples scene compute graph)
if(TARGET renderer::animation AND TARGET renderer::interaction)
    list(APPEND development_examples sprite)
endif()
foreach(example IN LISTS development_examples)
    add_executable(development_${example} ${example}_app.cpp)
    target_link_libraries(development_${example} PRIVATE renderer::opengl)
    set_target_properties(development_${example} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>"
        RENDERER_EXAMPLE_REQUIRES_OPENGL ON)
    if(MSVC)
        target_compile_options(development_${example} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(development_${example} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(TARGET development_sprite)
    target_link_libraries(development_sprite PRIVATE renderer::animation renderer::interaction)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Example tests are disabled.\n")
    endif()
endif()
