cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_features_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build feature integration example" ON)
    option(RENDERER_BUILD_TESTS "Register feature example checks" ON)
    option(RENDERER_GL_TESTS "Register OpenGL checks" OFF)
    option(RENDERER_VULKAN_TESTS "Register Vulkan checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Features example disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS ui text OPTIONAL_COMPONENTS opengl vulkan)
endif()
if(NOT TARGET renderer::opengl AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "Features example requires at least one graphics backend")
endif()
add_executable(renderer_features_demo main.cpp)
target_link_libraries(renderer_features_demo PRIVATE renderer::ui renderer::text)
foreach(backend IN ITEMS opengl vulkan)
    if(TARGET renderer::${backend})
        string(TOUPPER "${backend}" upper)
        target_compile_definitions(renderer_features_demo PRIVATE RENDERER_FEATURES_${upper}=1)
        target_link_libraries(renderer_features_demo PRIVATE renderer::${backend})
    endif()
endforeach()
target_compile_definitions(renderer_features_demo PRIVATE RENDERER_FEATURES_FONT="${renderer_TEXT_FONT}")
set_target_properties(renderer_features_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_features_demo PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(renderer_features_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        include("${renderer_EXAMPLE_TEST_HELPER}")
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Features tests disabled.\n")
    endif()
endif()
