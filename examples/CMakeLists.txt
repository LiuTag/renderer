cmake_minimum_required(VERSION 3.25)
# 新建目标统一归入解决方案 examples 文件夹，并由子目录继承。
set(CMAKE_FOLDER "examples")

# 在源码根工程中复用已有目标，也可以把安装后的 examples/ 作为独立 SDK 工程。
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_examples LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build examples from the installed SDK" ON)
    option(RENDERER_BUILD_TESTS "Register example runtime checks" ON)
    option(RENDERER_GL_TESTS "Register checks requiring an OpenGL display" OFF)
    option(RENDERER_VULKAN_TESTS "Register Vulkan example checks" OFF)
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core
        OPTIONAL_COMPONENTS opengl vulkan animation interaction audio tilemap collision particles environment presentation physics2d camera ui text)
endif()

if(RENDERER_BUILD_EXAMPLES)
    add_subdirectory(texture_text)
    if(TARGET renderer::opengl OR TARGET renderer::vulkan)
        if(TARGET renderer::text AND TARGET renderer::interaction)
            add_subdirectory(performance)
        endif()
        if(TARGET renderer::ui AND TARGET renderer::text)
            add_subdirectory(features)
        endif()
        if(TARGET renderer::text)
            add_subdirectory(text)
        endif()
        if(TARGET renderer::ui)
            add_subdirectory(ui)
        endif()
        add_subdirectory(offscreen)
        add_subdirectory(sprite_benchmark)
        if(TARGET renderer::animation AND TARGET renderer::interaction AND TARGET renderer::audio
                AND TARGET renderer::tilemap AND TARGET renderer::collision AND TARGET renderer::particles
                AND TARGET renderer::environment AND TARGET renderer::presentation AND TARGET renderer::physics2d AND TARGET renderer::camera AND TARGET renderer::ui AND TARGET renderer::text)
            add_subdirectory(game)
        endif()
        if(TARGET renderer::tilemap AND (TARGET renderer::collision OR TARGET renderer::physics2d) AND TARGET renderer::particles AND TARGET renderer::environment AND TARGET renderer::presentation)
            add_subdirectory(world)
        endif()
    endif()
    if(TARGET renderer::vulkan)
        add_executable(renderer_vulkan_demo demo.cpp)
        target_link_libraries(renderer_vulkan_demo PRIVATE renderer::vulkan)
        target_compile_definitions(renderer_vulkan_demo PRIVATE RENDERER_DEMO_VULKAN=1)
        set_target_properties(renderer_vulkan_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>" RENDERER_EXAMPLE_REQUIRES_VULKAN ON)
        if(MSVC)
            target_compile_options(renderer_vulkan_demo PRIVATE /W4 /permissive- /utf-8)
        endif()
        if(TARGET renderer::animation AND TARGET renderer::interaction)
            add_executable(renderer_vulkan_sprite_demo development/sprite_app.cpp)
            target_link_libraries(renderer_vulkan_sprite_demo PRIVATE renderer::vulkan renderer::animation renderer::interaction)
            target_compile_definitions(renderer_vulkan_sprite_demo PRIVATE RENDERER_DEMO_VULKAN=1)
            set_target_properties(renderer_vulkan_sprite_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>" RENDERER_EXAMPLE_REQUIRES_VULKAN ON)
            if(MSVC)
                target_compile_options(renderer_vulkan_sprite_demo PRIVATE /W4 /permissive- /utf-8)
            endif()
        endif()
    endif()
    if(TARGET renderer::opengl)
        add_executable(renderer_demo demo.cpp)
        target_link_libraries(renderer_demo PRIVATE renderer::opengl)
        set_target_properties(renderer_demo PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
        if(MSVC)
            target_compile_options(renderer_demo PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(renderer_demo PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        if(TARGET renderer::animation AND TARGET renderer::interaction)
            add_executable(renderer_sprite_demo development/sprite_app.cpp)
            target_link_libraries(renderer_sprite_demo PRIVATE
                renderer::opengl renderer::animation renderer::interaction)
            set_target_properties(renderer_sprite_demo PROPERTIES
                RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
            if(MSVC)
                target_compile_options(renderer_sprite_demo PRIVATE /W4 /permissive- /utf-8)
            else()
                target_compile_options(renderer_sprite_demo PRIVATE -Wall -Wextra -Wpedantic)
            endif()
        endif()
        add_subdirectory(development)
    endif()
    add_subdirectory(sdk_consumer)
    add_subdirectory(modules_consumer)
    if(TARGET renderer::audio)
        add_subdirectory(audio)
    endif()
    if(TARGET renderer::physics2d)
        add_subdirectory(physics2d)
    endif()
    if(TARGET renderer::camera)
        add_subdirectory(camera)
    endif()
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Example tests are disabled.\n")
    endif()
endif()
