cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_physics2d_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the Box2D integration example" ON)
    option(RENDERER_BUILD_TESTS "Register physics example checks" ON)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Physics example is disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS physics2d)
endif()
add_executable(renderer_physics2d_demo main.cpp)
target_link_libraries(renderer_physics2d_demo PRIVATE renderer::physics2d)
set_target_properties(renderer_physics2d_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_physics2d_demo PRIVATE /W4 /permissive- /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        add_test(NAME renderer.physics2d.example COMMAND renderer_physics2d_demo)
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Physics example tests are disabled.\n")
    endif()
endif()
