cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_offscreen_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the offscreen example" ON)
    option(RENDERER_BUILD_TESTS "Register offscreen example checks" ON)
    option(RENDERER_GL_TESTS "Enable OpenGL checks" OFF)
    option(RENDERER_VULKAN_TESTS "Enable Vulkan checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Offscreen example is disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS opengl vulkan)
endif()
if(NOT TARGET renderer::opengl AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "The offscreen example requires OpenGL or Vulkan")
endif()
add_executable(renderer_offscreen_demo main.cpp)
if(MINGW)
    target_link_options(renderer_offscreen_demo PRIVATE -municode)
endif()
foreach(backend IN ITEMS opengl vulkan)
    if(TARGET renderer::${backend})
        string(TOUPPER "${backend}" backend_upper)
        target_compile_definitions(renderer_offscreen_demo PRIVATE RENDERER_OFFSCREEN_${backend_upper}=1)
        target_link_libraries(renderer_offscreen_demo PRIVATE renderer::${backend})
    endif()
endforeach()
set_target_properties(renderer_offscreen_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_offscreen_demo PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(renderer_offscreen_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Offscreen tests are disabled.\n")
    endif()
endif()
