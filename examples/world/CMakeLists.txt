cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_world_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the modular world example" ON)
    option(RENDERER_BUILD_TESTS "Register world example checks" ON)
    option(RENDERER_GL_TESTS "Enable OpenGL checks" OFF)
    option(RENDERER_VULKAN_TESTS "Enable Vulkan checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# World example is disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core tilemap particles environment presentation OPTIONAL_COMPONENTS opengl vulkan collision physics2d)
endif()
if(NOT TARGET renderer::opengl AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "The world example requires an OpenGL or Vulkan component")
endif()
add_executable(renderer_world_demo main.cpp)
target_link_libraries(renderer_world_demo PRIVATE renderer::tilemap renderer::particles renderer::environment renderer::presentation)
if(TARGET renderer::physics2d)
    target_link_libraries(renderer_world_demo PRIVATE renderer::physics2d)
    target_compile_definitions(renderer_world_demo PRIVATE RENDERER_WORLD_PHYSICS2D=1)
elseif(TARGET renderer::collision)
    target_link_libraries(renderer_world_demo PRIVATE renderer::collision)
else()
    message(FATAL_ERROR "The world example requires physics2d or collision")
endif()
foreach(backend IN ITEMS opengl vulkan)
    if(TARGET renderer::${backend})
        string(TOUPPER "${backend}" backend_upper)
        target_compile_definitions(renderer_world_demo PRIVATE RENDERER_WORLD_${backend_upper}=1)
        target_link_libraries(renderer_world_demo PRIVATE renderer::${backend})
    endif()
endforeach()
set_target_properties(renderer_world_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_world_demo PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(renderer_world_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# World example tests are disabled.\n")
    endif()
endif()
