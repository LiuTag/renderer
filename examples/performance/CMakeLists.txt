cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_performance_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the optimization benchmark" ON)
    option(RENDERER_BUILD_TESTS "Register benchmark checks" ON)
    option(RENDERER_GL_TESTS "Register OpenGL checks" OFF)
    option(RENDERER_VULKAN_TESTS "Register Vulkan checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core text interaction OPTIONAL_COMPONENTS opengl vulkan)
endif()
if(NOT TARGET renderer::opengl AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "The performance example requires an OpenGL or Vulkan SDK component")
endif()
add_executable(renderer_performance main.cpp)
target_link_libraries(renderer_performance PRIVATE renderer::text renderer::interaction)
target_compile_definitions(renderer_performance PRIVATE RENDERER_PERFORMANCE_FONT="${renderer_TEXT_FONT}")
foreach(backend IN ITEMS opengl vulkan)
    if(TARGET renderer::${backend})
        string(TOUPPER "${backend}" upper)
        target_compile_definitions(renderer_performance PRIVATE RENDERER_PERFORMANCE_${upper}=1)
        target_link_libraries(renderer_performance PRIVATE renderer::${backend})
    endif()
endforeach()
set_target_properties(renderer_performance PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_performance PRIVATE /W4 /permissive- /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        include("${renderer_EXAMPLE_TEST_HELPER}")
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Tests disabled.\n")
    endif()
endif()
