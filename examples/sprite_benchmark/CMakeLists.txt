cmake_minimum_required(VERSION 3.25)
set(CMAKE_FOLDER "examples")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_sprite_benchmark_example LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    option(RENDERER_BUILD_EXAMPLES "Build the sprite benchmark" ON)
    option(RENDERER_BUILD_TESTS "Register benchmark smoke checks" ON)
    option(RENDERER_GL_TESTS "Register OpenGL benchmark checks" OFF)
    option(RENDERER_VULKAN_TESTS "Register Vulkan benchmark checks" OFF)
    if(NOT RENDERER_BUILD_EXAMPLES)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Examples are disabled.\n")
        return()
    endif()
    find_package(renderer 0.2 CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS opengl vulkan)
endif()

if(NOT TARGET renderer::opengl AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "The sprite benchmark requires an OpenGL or Vulkan renderer SDK component")
endif()
add_executable(renderer_sprite_benchmark main.cpp)
target_sources(renderer_sprite_benchmark PRIVATE ../common/forest_assets.hpp)
foreach(backend IN ITEMS opengl vulkan)
    if(TARGET renderer::${backend})
        target_link_libraries(renderer_sprite_benchmark PRIVATE renderer::${backend})
        string(TOUPPER "${backend}" backend_upper)
        target_compile_definitions(renderer_sprite_benchmark PRIVATE RENDERER_BENCHMARK_${backend_upper}=1)
    endif()
endforeach()
set_target_properties(renderer_sprite_benchmark PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
    target_compile_options(renderer_sprite_benchmark PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(renderer_sprite_benchmark PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(RENDERER_BUILD_TESTS)
        enable_testing()
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
            include("${CMAKE_CURRENT_LIST_DIR}/../../tests/example_tests.cmake")
        else()
            include("${renderer_EXAMPLE_TEST_HELPER}")
        endif()
        renderer_register_example_tests()
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestTestfile.cmake" "# Example tests are disabled.\n")
    endif()
endif()
