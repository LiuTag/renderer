# 单元与集成测试只在根工程 RENDERER_BUILD_TESTS=ON 时进入本目录。
set(CMAKE_FOLDER "tests")
if(RENDERER_VULKAN_TESTS AND NOT TARGET renderer::vulkan)
    message(FATAL_ERROR "RENDERER_VULKAN_TESTS requires RENDERER_BUILD_VULKAN=ON")
endif()
if(RENDERER_GL_TESTS AND NOT TARGET renderer::opengl)
    message(FATAL_ERROR "RENDERER_GL_TESTS requires RENDERER_BUILD_OPENGL=ON")
endif()

if(RENDERER_BUILD_TESTS)
    add_executable(renderer_sprite_tests sprite_tests.cpp)
    target_link_libraries(renderer_sprite_tests PRIVATE renderer_core)
    target_include_directories(renderer_sprite_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    renderer_warnings(renderer_sprite_tests)
    add_test(NAME renderer.sprites COMMAND renderer_sprite_tests)
    foreach(module IN ITEMS animation interaction audio)
        if(TARGET renderer_${module})
            add_executable(renderer_${module}_tests ${module}_tests.cpp)
            target_link_libraries(renderer_${module}_tests PRIVATE renderer_${module})
            target_include_directories(renderer_${module}_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
            renderer_warnings(renderer_${module}_tests)
            add_test(NAME renderer.${module} COMMAND renderer_${module}_tests)
        endif()
    endforeach()
    add_executable(renderer_tests core_tests.cpp)
    target_link_libraries(renderer_tests PRIVATE renderer_core)
    target_include_directories(renderer_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    renderer_warnings(renderer_tests)
    add_test(NAME renderer.core COMMAND renderer_tests)
    add_executable(renderer_engine_tests engine_tests.cpp)
    target_link_libraries(renderer_engine_tests PRIVATE renderer_core)
    target_include_directories(renderer_engine_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_definitions(renderer_engine_tests PRIVATE RENDERER_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    renderer_warnings(renderer_engine_tests)
    add_test(NAME renderer.engine COMMAND renderer_engine_tests)
    add_executable(renderer_event_tests event_tests.cpp)
    target_link_libraries(renderer_event_tests PRIVATE renderer_core)
    target_include_directories(renderer_event_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    renderer_warnings(renderer_event_tests)
    add_test(NAME renderer.events COMMAND renderer_event_tests)
endif()
if(RENDERER_GL_TESTS)
    add_executable(renderer_sprite_gl_tests sprite_gl_tests.cpp)
    target_link_libraries(renderer_sprite_gl_tests PRIVATE renderer_opengl)
    renderer_warnings(renderer_sprite_gl_tests)
    add_test(NAME renderer.opengl.sprites COMMAND renderer_sprite_gl_tests)
    set_tests_properties(renderer.opengl.sprites PROPERTIES TIMEOUT 120 LABELS "gpu")
    add_executable(renderer_window_tests window_tests.cpp)
    target_link_libraries(renderer_window_tests PRIVATE renderer_opengl glfw)
    target_include_directories(renderer_window_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_definitions(renderer_window_tests PRIVATE GLFW_INCLUDE_NONE)
    renderer_warnings(renderer_window_tests)
    add_test(NAME renderer.window COMMAND renderer_window_tests)
    set_tests_properties(renderer.window PROPERTIES TIMEOUT 60 LABELS "gpu")
    add_executable(renderer_fullscreen_tests fullscreen_tests.cpp)
    target_link_libraries(renderer_fullscreen_tests PRIVATE renderer_opengl glfw)
    target_include_directories(renderer_fullscreen_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_definitions(renderer_fullscreen_tests PRIVATE GLFW_INCLUDE_NONE)
    renderer_warnings(renderer_fullscreen_tests)
    add_test(NAME renderer.fullscreen COMMAND renderer_fullscreen_tests)
    set_tests_properties(renderer.fullscreen PROPERTIES TIMEOUT 60 LABELS "gpu" RUN_SERIAL TRUE)
    add_executable(renderer_gl_tests gl_tests.cpp)
    target_link_libraries(renderer_gl_tests PRIVATE renderer_opengl)
    renderer_warnings(renderer_gl_tests)
    add_test(NAME renderer.opengl COMMAND renderer_gl_tests)
    set_tests_properties(renderer.opengl PROPERTIES TIMEOUT 60 LABELS "gpu")
    add_executable(renderer_advanced_gl_tests advanced_gl_tests.cpp)
    target_link_libraries(renderer_advanced_gl_tests PRIVATE renderer_opengl)
    renderer_warnings(renderer_advanced_gl_tests)
    add_test(NAME renderer.opengl.advanced COMMAND renderer_advanced_gl_tests)
    set_tests_properties(renderer.opengl.advanced PROPERTIES TIMEOUT 120 LABELS "gpu")
    add_executable(renderer_aa_tests aa_gl_tests.cpp)
    target_link_libraries(renderer_aa_tests PRIVATE renderer_opengl)
    renderer_warnings(renderer_aa_tests)
    add_test(NAME renderer.opengl.aa COMMAND renderer_aa_tests)
    set_tests_properties(renderer.opengl.aa PROPERTIES TIMEOUT 120 LABELS "gpu")
endif()

# 保留既有可执行文件位置，方便直接运行与调试。
if(RENDERER_VULKAN_TESTS)
    if(TARGET renderer::opengl)
        add_executable(renderer_backend_parity_tests backend_parity_tests.cpp)
        target_link_libraries(renderer_backend_parity_tests PRIVATE renderer::opengl renderer::vulkan)
        renderer_warnings(renderer_backend_parity_tests)
        set_target_properties(renderer_backend_parity_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
        add_test(NAME renderer.vulkan.parity COMMAND renderer_backend_parity_tests)
        set_tests_properties(renderer.vulkan.parity PROPERTIES TIMEOUT 120 LABELS "gpu;vulkan")
    endif()
    add_executable(renderer_vulkan_window_tests vulkan_window_tests.cpp)
    target_link_libraries(renderer_vulkan_window_tests PRIVATE renderer::vulkan)
    renderer_warnings(renderer_vulkan_window_tests)
    set_target_properties(renderer_vulkan_window_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
    add_test(NAME renderer.vulkan.window COMMAND renderer_vulkan_window_tests)
    set_tests_properties(renderer.vulkan.window PROPERTIES TIMEOUT 120 LABELS "gpu;vulkan" RUN_SERIAL TRUE)
    foreach(suite IN ITEMS core advanced aa sprite)
        if(suite STREQUAL "core")
            set(test_source gl_tests.cpp)
        elseif(suite STREQUAL "advanced")
            set(test_source advanced_gl_tests.cpp)
        elseif(suite STREQUAL "aa")
            set(test_source aa_gl_tests.cpp)
        else()
            set(test_source sprite_gl_tests.cpp)
        endif()
        add_executable(renderer_vulkan_${suite}_tests ${test_source})
        target_link_libraries(renderer_vulkan_${suite}_tests PRIVATE renderer::vulkan)
        target_compile_definitions(renderer_vulkan_${suite}_tests PRIVATE RENDERER_TEST_VULKAN=1)
        renderer_warnings(renderer_vulkan_${suite}_tests)
        set_target_properties(renderer_vulkan_${suite}_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
        add_test(NAME renderer.vulkan.${suite} COMMAND renderer_vulkan_${suite}_tests)
        set_tests_properties(renderer.vulkan.${suite} PROPERTIES TIMEOUT 180 LABELS "gpu;vulkan")
    endforeach()
endif()

foreach(test_target IN ITEMS renderer_sprite_tests renderer_tests renderer_engine_tests
        renderer_event_tests renderer_animation_tests renderer_interaction_tests renderer_audio_tests
        renderer_sprite_gl_tests renderer_window_tests renderer_fullscreen_tests
        renderer_gl_tests renderer_advanced_gl_tests renderer_aa_tests)
    if(TARGET ${test_target})
        set_target_properties(${test_target} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
    endif()
endforeach()

include("${CMAKE_CURRENT_LIST_DIR}/example_tests.cmake")
renderer_register_example_tests()
